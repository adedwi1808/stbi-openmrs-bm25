\begin{frame}{BM25 pada MTSamples Sudah Diuji, tapi Sebagai Baseline Naif}
  \framesubtitle{Temuan baru: penelitian terkait belum menyasar optimasi tokenisasi BM25 untuk teks klinis.}
  \scriptsize
  \renewcommand{\arraystretch}{1.25}
  \begin{tabular}{p{0.30\textwidth}p{0.34\textwidth}p{0.28\textwidth}}
    \toprule
    \textbf{Studi} & \textbf{Fokus} & \textbf{Perlakuan BM25} \\
    \midrule
    Multi-Corpus Benchmark Study (Mikkelsen, 2026), termasuk korpus MTSamples ($n{=}500$) & Membandingkan model \textit{embedding} vs.\ BM25 untuk retrieval klinis & BM25 hanya sebagai \textit{baseline}: tokenisasi \textit{lowercase} + hapus tanda baca ($k_1{=}1{,}5$, $b{=}0{,}75$) \\
    Improved BM25 untuk \textit{Precision Medicine} (Zhang, \textit{BMC Med.\ Inform.\ Decis.\ Mak.}, 2021) & Optimasi bobot skor BM25 dengan \textit{co-word analysis} + Cuckoo Search & Perbaikan pada fungsi skor, bukan pada tahap tokenisasi/normalisasi teks \\
    \bottomrule
  \end{tabular}
  \vspace{0.28cm}
  \begin{block}{Gap yang diisi proyek ini}
    Belum ada studi yang menguji \textbf{tokenisasi khusus teks klinis} (mis.\ singkatan medis, negasi, istilah majemuk) untuk BM25 pada MTSamples, apalagi yang diintegrasikan langsung ke sistem rekam medis seperti OpenMRS.
  \end{block}
  \source{Sumber: Mikkelsen (2026) [2]; Zhang (2021) [3].}
\end{frame}

% 7
\begin{frame}{Tokenisasi Generik Berisiko Membuang Sinyal Klinis Penting}
  \framesubtitle{Urgensi diukur langsung pada korpus sendiri, bukan asumsi.}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.31\textwidth}
    \begin{block}{Negasi klinis}
      \textbf{87,7\%} dokumen memuat penanda negasi (``no'', ``denies'', ``without''). \textit{Stopword list} umum membuangnya, padahal maknanya berlawanan (``no fever'' vs.\ ``fever'').
    \end{block}
    \column{0.31\textwidth}
    \begin{block}{Istilah majemuk}
      \textbf{21,2\%} dokumen memuat frasa seperti ``shortness of breath''. \textit{Tokenizer} kata-tunggal memecahnya jadi token lepas, kehilangan makna sebagai satu konsep.
    \end{block}
    \column{0.31\textwidth}
    \begin{block}{Singkatan medis}
      \textbf{4,5\%} dokumen memuat singkatan seperti ``c/o'', ``DM'', ``HTN''. \textit{Tokenizer} generik memisahnya jadi token tak bermakna.
    \end{block}
  \end{columns}
  \vspace{0.15cm}
  \begin{block}{Kenapa ini diubah, bukan dipakai apa adanya}
    \scriptsize
    Baseline studi sebelumnya hanya \textit{lowercase} + hapus tanda baca: pada korpus ini itu membuang sinyal yang muncul di sebagian besar dokumen. Rencana: \textit{stopword list} klinis (pertahankan negasi) + \textit{stemming} istilah medis, dibandingkan terhadap baseline naif tersebut.
  \end{block}
  \source{Pengukuran pada \texttt{corpus.jsonl} (2.316 dokumen) [1]; rujukan pendukung: Gayen dkk.\ (2023) [4], Griffon dkk.\ (2012) [5].}
\end{frame}
