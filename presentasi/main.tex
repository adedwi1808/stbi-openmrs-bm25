  \begin{columns}[T,onlytextwidth]
    \small
    \column{0.48\textwidth}
    \begin{block}{Masalah \& tujuan}
      \begin{itemize}
        \item Pencarian RME \textit{exact-match} gagal menangani sinonim, singkatan, dan variasi terminologi medis.
        \item Tujuan: modul IR berbasis BM25 di OpenMRS yang menangkap relevansi semantik teks klinis.

      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Kontribusi}
      \begin{itemize}
        \item Bukan sekadar ``BM25 di dataset medis'': studi ablasi lapisan normalisasi + ekspansi klinis.
        \item 5 varian retrieval (V0-V4) dibandingkan berdampingan.
        \item Modul OpenMRS + endpoint JSON + UI dropdown, teruji berjalan.
        \item Evaluasi TREC-style: P@K, Recall@K, nDCG@K.
      \end{itemize}
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Urgensi Diukur Langsung pada Dataset (Bukan Asumsi)}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.31\textwidth}
    \begin{block}{Negasi klinis}
      88,0\% dokumen (13 term). \textit{Stopword list} umum membuang ``no/denies/without'', padahal maknanya berlawanan.
    \end{block}
    \column{0.31\textwidth}
    \begin{block}{Istilah majemuk}
      60,9\% dokumen (28 frasa). \textit{Tokenizer} kata-tunggal memecah ``shortness of breath'' jadi token lepas.
    \end{block}
    \column{0.31\textwidth}
    \begin{block}{Singkatan medis}
      28,8\% dokumen (28 singkatan). ``c/o'', ``DM'', ``HTN'', ``CKD'' tak bermakna bila dipecah generik.
    \end{block}
  \end{columns}
  \vspace{0.15cm}
  \begin{block}{Implikasi}
    \scriptsize
    Baseline sebelumnya (lowercase + hapus tanda baca) membuang sinyal yang muncul di sebagian besar dokumen. Solusi: normalisasi klinis (negasi dipertahankan, majemuk dikolaps, singkatan diekspansi) + ekspansi sinonim, dibandingkan ablation terhadap baseline naif.
  \end{block}
  \source{Pengukuran pada \texttt{corpus.jsonl} (2.316 dokumen) via \texttt{tools/build\_lexicon.py}.}
\end{frame}

% 9b
\begin{frame}{Penjelasan Tiap Teknik}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \textbf{\color{ugmBlueDark}Negasi dipertahankan (V1)}\\
    \scriptsize
    Penanda seperti \texttt{no}, \texttt{denies}, \texttt{without} biasanya dibuang \textit{stopword list} umum. Di sini dipertahankan agar makna klinis tidak terbalik (mis. ``no fever'' $\neq$ ``fever'').

    \vspace{0.35cm}
    \normalsize
    \textbf{\color{ugmBlueDark}Istilah majemuk dikolaps (V1)}\\
    \scriptsize
    Frasa seperti ``shortness of breath'' diperlakukan sebagai satu unit makna, bukan dipecah menjadi token lepas oleh \textit{tokenizer} kata-tunggal.

    \vspace{0.35cm}
    \normalsize
    \textbf{\color{ugmBlueDark}Ekspansi singkatan (V1)}\\
    \scriptsize
    Singkatan medis (\texttt{DM}, \texttt{HTN}, \texttt{CKD}, dst.) diekspansi dua arah, saat \textit{indexing} dan saat \textit{query}, sehingga bentuk singkat dan panjang saling ditemukan.

    \column{0.48\textwidth}
    \textbf{\color{ugmBlueDark}Ekspansi sinonim (V2)}\\
    \scriptsize
    Istilah awam pengguna (mis. ``high blood pressure'') diperluas ke istilah medis (mis. ``hypertension'') hanya di sisi \textit{query}, tanpa mengubah index.

    \vspace{0.35cm}
    \normalsize
    \textbf{\color{ugmBlueDark}Char n-gram (V3)}\\
    \scriptsize
    Dokumen diindeks berbasis potongan karakter (3-5 huruf), bukan kata utuh, sehingga tahan terhadap salah ketik dan variasi morfologi tanpa bergantung pada \textit{lexicon}.

    \vspace{0.35cm}
    \normalsize
    \textbf{\color{ugmBlueDark}Pseudo-relevance feedback (V4)}\\
    \scriptsize
    Ekspansi query otomatis: term yang sering muncul pada dokumen teratas hasil pencarian awal ditambahkan ke query, tanpa kamus manual.
  \end{columns}
\end{frame}

\begin{frame}{Hasil Evaluasi: Normalisasi Klinis Meningkatkan nDCG}
  \centering
  \renewcommand{\arraystretch}{1.25}
  \begin{tabular}{|l|c|c|c|}
    \hline
    \textbf{Varian}              & \textbf{P@10}     & \textbf{Recall@10} & \textbf{nDCG@10}  \\
    \hline
    V0 · Naive BM25              & 0{,}5400          & \textbf{0{,}4312}  & 0{,}6832          \\
    \hline
    V1 · Normalisasi klinis      & \textbf{0{,}5420} & 0{,}4284           & \textbf{0{,}6865} \\
    \hline
    V2 · Klinis + ekspansi       & 0{,}5120          & 0{,}4192           & 0{,}6509          \\
    \hline
    V3 · Char n-gram             & 0{,}5280          & 0{,}4055           & 0{,}6446          \\
    \hline
    V4 · Klinis + ekspansi + PRF & 0{,}5020          & 0{,}4139           & 0{,}6186          \\
    \hline
  \end{tabular}
  \vspace{0.25cm}
  \begin{columns}[T,onlytextwidth]
    \small
    \column{0.48\textwidth}
    \begin{block}{Temuan}
      V1 $>$ V0 pada P@10 \& nDCG@10 $\Rightarrow$ normalisasi klinis memberi perbaikan kecil namun konsisten. V2/V4 turun presisi (trade-off ekspansi), dilaporkan apa adanya.
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Verifikasi fungsional}
      Query ``CKD'' $\rightarrow$ menemukan ``Chronic Kidney Disease - Followup''. Query ``hypertension'' $\rightarrow$ menemukan dokumen ber-``HTN''.
    \end{block}
  \end{columns}
  \source{Sumber: \texttt{eval/results.txt}, dijalankan via \texttt{EvalRunner}.}
\end{frame}

% 16
\begin{frame}{Kesimpulan}
  \begin{block}{Apa yang dicapai}
    \begin{itemize}
      \item Mesin IR BM25 dengan lapisan normalisasi + ekspansi klinis, terintegrasi ke OpenMRS sebagai modul.
      \item Ablation study 5 varian pada dataset MTSamples (2.316 dokumen).
      \item Validasi berbasis data: negasi/majemuk/singkatan muncul di mayoritas dokumen.
      \item Hasil: V1 (normalisasi klinis) unggul tipis atas baseline naif, trade-off presisi pada varian ekspansi dilaporkan jujur.
      \item Purwarupa pencarian cerdas yang siap dipakai dan dapat diperluas.
    \end{itemize}
  \end{block}
\end{frame}
